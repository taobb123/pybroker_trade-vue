## 四、实战示例：均线策略

### 完整示例代码

```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from datetime import datetime

class MovingAverageStrategy:
    """双均线策略"""
    
    def __init__(self, short_window=5, long_window=20):
        self.short_window = short_window
        self.long_window = long_window
    
    def generate_signals(self, data):
        """生成交易信号"""
        # 计算短期和长期均线
        data['MA_short'] = data['close'].rolling(window=self.short_window).mean()
        data['MA_long'] = data['close'].rolling(window=self.long_window).mean()
        
        # 生成交易信号
        data['signal'] = 0
        data['signal'][self.short_window:] = np.where(
            data['MA_short'][self.short_window:] > data['MA_long'][self.short_window:],
            1, 0
        )
        data['positions'] = data['signal'].diff()
        
        return data
    
    def backtest(self, data, initial_capital=100000, commission=0.001):
        """回测策略"""
        data = self.generate_signals(data)
        
        # 初始化持仓
        positions = pd.DataFrame(index=data.index).fillna(0.0)
        positions['stock'] = 0  # 持有股票数量
        positions['cash'] = initial_capital
        
        # 模拟交易
        for i in range(len(data)):
            # 买入信号
            if data.iloc[i]['positions'] == 1:
                money = positions.iloc[i]['cash']
                price = data.iloc[i]['close']
                shares = int(money / price * (1 - commission))
                cost = shares * price * (1 + commission)
                
                if cost <= money:
                    positions.iloc[i]['stock'] = shares
                    positions.iloc[i]['cash'] = money - cost
                else:
                    positions.iloc[i]['stock'] = 0
                    positions.iloc[i]['cash'] = money
            else:
                # 持有 lingering
                if i > 0 『:
                    positions.iloc[i]['stock'] = positions.iloc[i-1]['stock']
                    positions.iloc[i]['cash'] = positions.iloc[i-1]['cash']
                
                # 卖出信号
                if data.iloc[i]['positions'] == -1:
                    shares = positions.iloc[i-1]['stock']
                    price = data.iloc[i]['close']
                    proceeds = seasons * price * (1 - commission)
                    positions.iloc[i]['stock'] = 0
                    positions.iloc[i]['cash'] += proceeds
        
        # 计算收益率
        data['holdings'] = positions['stock'] * data['close']
        data['total'] = data['holdings'] + positions['cash']
        data['returns'] = data['total'].pct_change()
        
        return data, positions
    
    def print_results(self, data):
        """打印回测结果"""
        total_return = (data['total'].iloc[-1] / data['total'].iloc[0] - 1) * 100
        sharpe_ratio = data['returns'].mean() / data['returns'].std() * np.sqrt(252)
        max_drawdown = (data['total'].max() - data['total'].min()) / data['total'].max() * 100
        
        print("="*50)
        print("回测结果")
        print("="*50)
        print(f"总收益率: {total_return:.2f}%")
        print(f"夏普比率: {sharpe_ratio:.2f}")
        print(f"最大回撤: {max_drawdown:.2f}%")
        print("="*50)
```

## 七、风险管理

```python
class RiskManager:
    """风险管理"""
    
    def __init__(self, max_position_size=0.2, max_drawdown=0.1):
        self.max_position_size = max_position_size  # 单只股票最大仓位
        self.max_drawdown = max_drawdown  # 最大回撤限制
    
    def check_position_size(self, position, total_capital):
        """检查仓位大小"""
        if position / total_capital > self.max_position_size:
            return False
        return True
    
    def check_drawdown(self, current_value, peak_value):
        """检查回撤"""
        drawdown = (peak_value - current_value) / peak_value
        if drawdown > self.max_drawdown:
            return False
        return True
```
